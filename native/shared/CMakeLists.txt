cmake_minimum_required(VERSION 3.22)
project(poleline LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(poleline_core STATIC poleline.cpp)
if(MSVC)
    target_compile_options(poleline_core PRIVATE /W4 /WX /fp:strict)
else()
    target_compile_options(poleline_core PRIVATE -Wall -Wextra -Werror -ffp-contract=off -fno-fast-math)
endif()
set_target_properties(poleline_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(ANDROID)
    add_library(poleline SHARED android_jni.cpp)
    target_link_libraries(poleline PRIVATE poleline_core)
else()
    enable_testing()
    add_test(NAME offline_gameplay COMMAND poleline_tests)
endif()
add_executable(poleline_tests tests.cpp)
target_link_libraries(poleline_tests PRIVATE poleline_core)
